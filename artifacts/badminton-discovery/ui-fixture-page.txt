'use client';
import React,{useState,useEffect} from 'react';
import {DiscoveryScoutModal} from '@/components/sport-hub/discovery-scout-modal';
import preview from '../../../../artifacts/badminton-discovery/ui-preview.json';
export default function QA(){const[open,setOpen]=useState(false);const[count,setCount]=useState(5);const[confirmed,setConfirmed]=useState(0);const[fail,setFail]=useState(false);
useEffect(()=>{const original=window.fetch;window.fetch=async(url,init)=>{if(String(url)==='/api/sport-hub/scout'&&init?.method==='POST'){const body=JSON.parse(String(init.body));if(body.action==='confirm'){if(fail)return Response.json({success:false,error:'QA import failure'},{status:500});if(body.selected.length!==count||body.selected.some((c:any)=>!c.locationConfirmed))return Response.json({success:false,error:'QA invalid selection'},{status:400});setConfirmed(body.selected.length);return Response.json({success:true,summary:`QA imported ${body.selected.length} profiles`});}return Response.json({...preview,candidates:preview.candidates.slice(0,count),newCount:Math.max(count-1,0),partial:count<5,diagnostics:{...preview.diagnostics,providerCount:count<5?count:15,returnedCount:count},warnings:count<5?[`Only ${count} of 5 requested profiles passed discovery checks.`]:[]});}return original(url,init)};return()=>{window.fetch=original}},[count,fail]);
return <main className="p-6"><h1>Discovery QA — recorded Apify dataset, mocked API and persistence</h1><label>Fixture result count <select aria-label="Fixture result count" value={count} onChange={e=>setCount(Number(e.target.value))}><option value={5}>5</option><option value={1}>1</option><option value={0}>0</option></select></label><label><input type="checkbox" checked={fail} onChange={e=>setFail(e.target.checked)}/>Simulate import failure</label><button onClick={()=>setOpen(true)}>Open Discovery QA</button><p>QA confirmed selections: {confirmed}</p><DiscoveryScoutModal isOpen={open} onClose={()=>setOpen(false)}/></main>}
